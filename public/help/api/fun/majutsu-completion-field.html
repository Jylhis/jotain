;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-completion.el
(defun majutsu-completion-field (text &optional face)
  "Return TEXT formatted as one annotation field."
  (when (and text (not (string-empty-p text)))
    (if face
        (propertize (format "%s" text) 'face face)
      (format "%s" text))))
