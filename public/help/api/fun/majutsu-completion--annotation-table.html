(defun majutsu-completion--annotation-table (items)
  "Return candidate annotation table for ITEMS, or nil."
  (let (annotations)
    (dolist (item items)
      (when (consp item)
        (unless annotations
          (setq annotations (make-hash-table :test #'equal)))
        (puthash (car item) (cdr item) annotations)))
    annotations))
