(defun kkp--disable-in-active-terminals()
  "In all terminals with active KKP, pop the previously pushed enhancement flag."
  (dolist (terminal (terminal-list))
    (when (kkp--active-p terminal)
      (kkp--terminal-teardown terminal))))
