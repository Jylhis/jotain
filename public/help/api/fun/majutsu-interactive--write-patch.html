Source Code

;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-interactive.el
(defun majutsu-interactive--write-patch (patch)
  "Write PATCH to a temporary file and return its path."
  (let ((file (expand-file-name "patch.diff" (majutsu-interactive--temp-dir))))
    (with-temp-file file
      (insert patch))
    file))
