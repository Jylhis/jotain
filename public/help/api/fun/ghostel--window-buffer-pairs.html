Function: ghostel--window-buffer-pairs

(ghostel--window-buffer-pairs WINDOWS)

Source Code

;; Defined in /nix/store/1jkgkg7y6y6hp0xia3dsmcl5zaj0bnhq-emacs-packages-deps/share/emacs/site-lisp/ghostel.el
(defun ghostel--window-buffer-pairs (windows)
  "Return (WINDOW . BUFFER) pairs for WINDOWS."
  (mapcar (lambda (window) (cons window (window-buffer window))) windows))
