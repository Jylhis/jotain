(defun tempel--user-element (elt)
  "Evaluate user element ELT."
  (run-hook-wrapped 'tempel-user-elements
                    (lambda (hook elt fields)
                      (condition-case nil
                          (funcall hook elt)
                        (wrong-number-of-arguments
                         (funcall hook elt fields))))
                    elt (cdar tempel--active)))
