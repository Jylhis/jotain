(defun eca-mcp--parse-env-lines (lines)
  "Parse LINES (each \"KEY=VALUE\") into a plist for JSON encoding."
  (let (out)
    (dolist (l lines)
      (when-let* ((pos (string-match "=" l)))
        (let ((k (substring l 0 pos))
              (v (substring l (1+ pos))))
          (push v out)
          (push (intern (concat ":" k)) out))))
    (nreverse out)))
