(defun cape--case-replace-list (flag input strs)
  "Replace case of STRS depending on INPUT and FLAG."
  (if (if (eq flag 'case-replace) case-replace flag)
      (mapcar (apply-partially #'cape--case-replace flag input) strs)
    strs))
