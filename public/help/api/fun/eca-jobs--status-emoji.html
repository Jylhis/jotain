(defun eca-jobs--status-emoji (status)
  "Return a colored emoji circle for STATUS."
  (pcase status
    ("running" "🟡")
    ("completed" "✅")
    ("failed" "🔴")
    ("killed" "⚫")
    (_ "⚫")))
