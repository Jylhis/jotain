(defun haskell-cabal-find-subsection (section name)
  "Find subsection with name NAME"
  (let ((downcase-name (downcase name)))
    (haskell-cabal-find-subsection-by
     section
     `(lambda (subsection)
        (string= (downcase (haskell-cabal-section-name subsection))
                 ,downcase-name)))))
