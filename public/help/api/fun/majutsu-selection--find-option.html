(defun majutsu-selection--find-option (id)
  (when (boundp 'transient--suffixes)
    (seq-find (lambda (obj)
                (and (cl-typep obj 'majutsu-selection-option)
                     (equal (majutsu-selection--selection-id obj) id)))
              transient--suffixes)))
