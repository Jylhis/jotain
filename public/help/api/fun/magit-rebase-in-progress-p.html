(defun magit-rebase-in-progress-p ()
  "Return t if a rebase is in progress."
  (let ((dir (magit-gitdir)))
    (or (file-exists-p (expand-file-name "rebase-merge" dir))
        (file-exists-p (expand-file-name "rebase-apply/onto" dir)))))
