(defun ghostel--range-overlaps-p (beg end ranges)
  "Non-nil when BEG..END overlaps any (START . STOP) cons in RANGES."
  (and (seq-find (lambda (range)
                   (and (< beg (cdr range)) (> end (car range))))
                 ranges)
       t))
