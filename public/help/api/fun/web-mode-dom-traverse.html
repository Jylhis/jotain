(defun web-mode-dom-traverse ()
  "Traverse html dom tree."
  (interactive)
  (cond
    ((web-mode-element-child)
     )
    ((web-mode-element-sibling-next)
     )
    ((and (web-mode-element-parent)
          (not (web-mode-element-sibling-next)))
     (goto-char (point-min)))
    (t
     (goto-char (point-min)))
    ) ;cond
  )
