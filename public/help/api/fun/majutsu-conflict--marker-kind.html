;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-conflict.el
(defun majutsu-conflict--marker-kind (char)
  "Return semantic marker kind for marker CHAR."
  (pcase char
    (?< 'begin)
    (?> 'end)
    (?+ 'add)
    (?- 'remove)
    (?% 'diff)
    (?\\ 'note)
    (?| 'git-ancestor)
    (?= 'git-separator)
    (_ nil)))
