(defun denote-sluggify-keywords-and-apply-rules (keywords)
  "Sluggify KEYWORDS, which is a list of strings."
  (mapcar (lambda (keyword)
            (denote-sluggify-and-apply-rules 'keyword keyword))
          keywords))
