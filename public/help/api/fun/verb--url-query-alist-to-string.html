(defun verb--url-query-alist-to-string (query)
  "Return alist query string QUERY as a string."
  (when query
    (mapconcat (lambda (kv)
                 (if (cdr kv)
                     (concat (car kv) "=" (cdr kv))
                   (car kv)))
               query
               "&")))
