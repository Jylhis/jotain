;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-file.el
(defun majutsu-file--summary-status-label (code)
  "Return a human-readable status label for jj diff summary CODE."
  (pcase code
    ("M" "modified")
    ("A" "added")
    ("D" "deleted")
    ("R" "renamed")
    ("C" "copied")
    (_ code)))
