(defun tagref-list--make-entries (tags)
  "Convert TAGS alist to tabulated-list entries."
  (mapcar (lambda (tag)
            (let ((name (car tag))
                  (file (cadr tag))
                  (line (cddr tag)))
              (list name (vector name file (number-to-string line)))))
          tags))
