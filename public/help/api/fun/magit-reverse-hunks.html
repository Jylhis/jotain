(defun magit-reverse-hunks (sections args)
  (magit-confirm 'reverse
    (list "Reverse %d hunks from %s"
          (length sections)
          (magit-section-parent-value (car sections))))
  (magit-reverse-apply sections #'magit-apply-hunks args))
