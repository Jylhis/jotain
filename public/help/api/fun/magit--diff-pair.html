(defun magit--diff-pair (buffer a b &optional file)
  (with-temp-buffer
    (insert (format ":100644 100644 %s %s M\0%s\0" a b (or file "blob")))
    (call-process-region (point-min) (point-max)
                         (magit-git-executable) nil buffer nil
                         "diff-pairs" "-z")))
