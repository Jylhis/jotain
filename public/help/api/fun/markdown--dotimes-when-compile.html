(defmacro markdown--dotimes-when-compile (i-n body)
  (declare (indent 1) (debug ((symbolp form) form)))
  (let ((var (car i-n))
        (n (cadr i-n))
        (code ()))
    (dotimes (i (eval n t))
      (push (eval body `((,var . ,i))) code))
    `(progn ,@(nreverse code))))
