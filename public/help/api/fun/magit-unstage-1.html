(defun magit-unstage-1 (files)
  (magit-run-before-change-functions files "unstage")
  (if (magit-no-commit-p)
      (magit-run-git "rm" "--cached" "--" files)
    (magit-run-git "reset" "HEAD" "--" files))
  (magit-run-after-apply-functions files "unstage"))
