(defun tagref--get-tags ()
  "Get all tags from the project.
Returns an alist of (NAME . (FILE . LINE-NUMBER))."
  (when-let ((output (tagref--call-process "list-tags")))
    (delq nil (mapcar #'tagref--parse-tag-line
                      (split-string output "\n" t)))))
