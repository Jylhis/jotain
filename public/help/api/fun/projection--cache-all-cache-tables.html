(defun projection--cache-all-cache-tables ()
  "Determine the list of all known cache tables."
  (seq-uniq
   (append
    (list 'projection--project-cache)
    (mapcar
     (lambda (cache-config)
       (plist-get (cdr cache-config) :cache-var))
     projection--cache-var-alist))))
