Function: majutsu--regexp-char-class

(majutsu--regexp-char-class CHARS)

Documentation

Source Code

;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-jj.el
(defun majutsu--regexp-char-class (chars)
  "Return CHARS escaped for use inside a regexp character class."
  (mapconcat (lambda (ch)
               (if (memq ch '(?\[ ?\] ?\\ ?^ ?-))
                   (concat "\\" (char-to-string ch))
                 (char-to-string ch)))
             chars
             ""))
