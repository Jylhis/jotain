Function: combobulate--kill-nodes

(combobulate--kill-nodes NODES)

;; Defined in /nix/store/9y0jasyzsqjcbfkwqsgdyxcnskjnsypg-emacs-packages-deps/share/emacs/site-lisp/combobulate-manipulation.el
(defun combobulate--kill-nodes (nodes)
  "Kill between the smallest and greatest range of NODES."
  (if-let (bounds (combobulate-node-range-extent nodes))
      (kill-region (car bounds) (cdr bounds))
    (error "No nodes to kill")))
