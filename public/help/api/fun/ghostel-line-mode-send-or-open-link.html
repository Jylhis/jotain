(defun ghostel-line-mode-send-or-open-link ()
  "Follow the link at point, or send the current line-mode input."
  (interactive)
  (let ((url (ghostel--uri-at-pos (point))))
    (if url
        (ghostel--open-link url)
      (ghostel-line-mode-send))))
