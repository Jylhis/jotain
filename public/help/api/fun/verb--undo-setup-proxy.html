(defun verb--undo-setup-proxy (rs)
  "Undo any HTTP proxy configuration specified by RS."
  (when-let ((proxy (verb--request-spec-metadata-get rs "proxy")))
    (setq url-proxy-services (delete (cons "http" proxy) url-proxy-services))))
