(defun gnuplot--end-of-command ()
  "Move point to end of command containing point."
  (let ((limit (gnuplot--point-at-end-of-continuation)))
    (while
        (and
         (search-forward ";" limit 'lim)
         (gnuplot--in-string-or-comment)))
    (skip-chars-backward ";")
    (skip-syntax-backward "-")))
