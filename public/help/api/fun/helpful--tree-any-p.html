(defun helpful--tree-any-p (pred tree)
  "Walk TREE, applying PRED to every subtree.
Return t if PRED ever returns t."
  (catch 'found
    (let ((stack (list tree)))
      (while stack
        (let ((next (pop stack)))
          (cond
           ((funcall pred next)
            (throw 'found t))
           ((consp next)
            (push (car next) stack)
            (push (cdr next) stack))))))
    nil))
