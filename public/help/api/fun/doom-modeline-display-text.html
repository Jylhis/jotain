(defun doom-modeline-display-text (text)
  "Display TEXT in mode-line."
  (let ((text (string-replace "%" "%%" (or text ""))))
    (if (doom-modeline--active)
        text
      (doom-modeline-propertize-text text 'mode-line-inactive))))
