(defun embark--record-history-p (action)
  "Return non-nil if ACTION should push the current target to history."
  (pcase embark-record-minibuffer-history
    ('t t)
    ('nil nil)
    (`(skip . ,actions) (not (memq action actions)))
    (_ nil)))
