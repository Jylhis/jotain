(defun eca-workspaces--session-cost (session)
  "Return the total cost of all SESSION chats, or nil."
  (let ((total (-sum (-map (lambda (buffer)
                             (string-to-number
                              (or (buffer-local-value 'eca-chat--session-cost
                                                      buffer)
                                  "0")))
                           (eca-chat-buffers session)))))
    (when (> total 0) total)))
