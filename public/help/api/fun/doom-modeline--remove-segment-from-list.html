(defun doom-modeline--remove-segment-from-list (list segment)
  "Remove SEGMENT from LIST.
Returns the modified list."
  (cond ((null list) nil)
        ((eq (car list) segment)
         (doom-modeline--remove-segment-from-list (cdr list) segment))
        (t
         (cons (car list)
               (doom-modeline--remove-segment-from-list (cdr list) segment)))))
