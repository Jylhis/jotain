;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-edit.el
(defun majutsu-edit--build-diffedit-args (from to file)
  "Build `jj diffedit' arguments for FROM, TO and FILE."
  (append (cond
           ((and from to)
            (list "--from" from "--to" to))
           (to
            (list "-r" to))
           (from
            (list "-r" from))
           (t
            (list "-r" "@")))
          (when file
            (list "--" file))))
