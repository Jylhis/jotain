(defun dape-breakpoint-toggle ()
  "Add or remove breakpoint at current line."
  (interactive)
  (if (cl-member nil (dape--breakpoints-at-point)
                 :key #'dape--source-breakpoint-type)
      (dape-breakpoint-remove-at-point)
    (dape--breakpoint-place)))
