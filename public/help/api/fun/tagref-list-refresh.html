(defun tagref-list-refresh ()
  "Refresh the tag listing."
  (interactive)
  (let ((tags (tagref--get-tags)))
    (setq tagref-list--tags tags)
    (setq tabulated-list-entries (tagref-list--make-entries tags))
    (tabulated-list-print t)))
