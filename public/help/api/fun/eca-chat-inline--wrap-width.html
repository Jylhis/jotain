(defun eca-chat-inline--wrap-width (ov)
  "Return the column to wrap OV's answer at."
  (or eca-chat-inline-wrap-column
      (let ((win (get-buffer-window (overlay-buffer ov))))
        (max 20 (- (if win (window-body-width win) 80) 2)))))
