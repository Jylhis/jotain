;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-log.el
(defun majutsu-log--cleanup-copied-string (string)
  "Strip Majutsu log UI properties from copied STRING."
  (when (stringp string)
    (remove-list-of-text-properties
     0 (length string)
     '(majutsu-log-module
       majutsu-log-field
       majutsu-log-column
       majutsu-log-entry-id
       majutsu-log-decoration
       majutsu-log-tail-spacer
       line-prefix
       wrap-prefix
       display)
     string))
  string)
