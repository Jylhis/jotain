;;;###autoload
(defun avy-isearch ()
  "Jump to one of the current isearch candidates."
  (interactive)
  (avy-with avy-isearch
    (let ((avy-background nil)
          (avy-case-fold-search case-fold-search))
      (prog1
          (avy-process
           (avy--regex-candidates (if isearch-regexp
                                      isearch-string
                                    (regexp-quote isearch-string))))
        (isearch-done)))))
