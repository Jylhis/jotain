(defun eca-chat-inline--start-turn (ov)
  "Reset OV to mirror a fresh turn."
  (overlay-put ov 'eca-chat-inline--text-acc "")
  (overlay-put ov 'eca-chat-inline--scroll nil)
  (eca-chat-inline--set-status ov "Waiting model..." 'running))
