(defun projection-golang-list-artifacts ()
  "List go packages as artifacts."
  (cl-loop
   for (short-name . package) in (projection-golang--list-packages)
   collect `((name . ,short-name)
             (type . go-package)
             (category . "Go package")
             (go-package . ,package)
             (debuggable . t))))
