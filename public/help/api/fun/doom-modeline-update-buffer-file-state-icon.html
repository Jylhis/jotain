;; Defined in /nix/store/mxkjlhf33q1xxjiii3w5v8nn7avms0p9-emacs-packages-deps/share/emacs/site-lisp/elpa/doom-modeline-20260828.1146/doom-modeline-segments.el
(defun doom-modeline-update-buffer-file-state-icon (&rest _)
  "Update the buffer or file state in mode-line."
  (when doom-modeline-buffer-state-icon
    (ignore-errors
      (concat
       (cond (buffer-read-only
              (doom-modeline-buffer-file-state-icon
               "nf-md-lock" "🔒" "%1*"
               'doom-modeline-warning))
             ((and doom-modeline-buffer-modification-icon
                   buffer-file-name
                   (buffer-modified-p))
              (doom-modeline-buffer-file-state-icon
               "nf-md-content_save_edit" "💾" "%1*"
               'doom-modeline-warning))
             ((and buffer-file-name
                   ;; Avoid freezing while connection is lost
                   (not (file-remote-p buffer-file-name))
                   (not (file-exists-p buffer-file-name)))
              (doom-modeline-buffer-file-state-icon
               "nf-md-cancel" "🚫" "!"
               'doom-modeline-urgent))
             ((not (or (and buffer-file-name
                            (file-remote-p buffer-file-name))
                       (verify-visited-file-modtime (current-buffer))))
              (doom-modeline-buffer-file-state-icon
               "nf-md-reload_alert" "⟳" "%1*"
               'doom-modeline-warning))
             (t ""))
       (when (or (buffer-narrowed-p)
                 (and (bound-and-true-p fancy-narrow-mode)
                      (fancy-narrow-active-p))
                 (bound-and-true-p dired-narrow-mode))
         (doom-modeline-buffer-file-state-icon
          "nf-md-unfold_less_horizontal" "↕" "><"
          'doom-modeline-warning))))))
