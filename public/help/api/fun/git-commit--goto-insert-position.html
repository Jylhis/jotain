(defun git-commit--goto-insert-position (bound)
  (let ((match (point)))
    (cond ((re-search-backward (format "^%s -+ >8 -+" comment-start) nil t))
          ((and (eobp) (bolp))
           (forward-line -1)))
    (while (and (or (looking-at comment-start)
                    (looking-at "[\s\t]*$"))
                (or (not bound) (> (point) bound))
                (not (bobp)))
      (setq match (point))
      (forward-line -1))
    (goto-char match)))
