(defun go-tag--parse-option (tags)
  (mapconcat 'identity
             (delete
              ""
              (split-string
               (mapconcat
                (lambda(str-lst)
                  (when (< 1 (length str-lst))
                    (concat (car str-lst) "=" (cadr str-lst))))
                (mapcar
                 (lambda(str)
                   (split-string str ","))
                 (split-string tags))
                ",")
               ","))
             ","))
