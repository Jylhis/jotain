Paste FILES into the terminal as shell-quoted paths, followed by a space.

(defun ghostel--dnd-send-files (files)
  "Paste FILES into the terminal as shell-quoted paths, followed by a space."
  (ghostel-paste-string
   (concat (mapconcat #'shell-quote-argument files " ") " ")))
