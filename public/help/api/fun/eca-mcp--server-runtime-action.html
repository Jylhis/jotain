(defun eca-mcp--server-runtime-action (server)
  "Return the runtime action available for MCP SERVER."
  (pcase (plist-get server :status)
    ((or "running" "starting") 'stop)
    ((or "stopped" "failed" "disabled") 'start)
    ("requires-auth" 'connect)
    (_ nil)))
