(defun claude-code-ide--get-session-context ()
  "Get current session context for logging."
  (if claude-code-ide-log-with-context
      (format "[%s]" (or (ignore-errors
                           (file-name-nondirectory
                            (directory-file-name
                             (project-root (project-current)))))
                         "no-project"))
    ""))
