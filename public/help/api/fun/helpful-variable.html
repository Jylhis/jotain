;;;###autoload
(defun helpful-variable (symbol)
  "Show help for variable named SYMBOL."
  (interactive
   (list (helpful--read-symbol
          "Variable: "
          (helpful--variable-at-point)
          #'helpful--variable-p)))
  (helpful--update-and-switch-buffer symbol nil))
