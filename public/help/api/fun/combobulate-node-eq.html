;; Defined in /nix/store/9y0jasyzsqjcbfkwqsgdyxcnskjnsypg-emacs-packages-deps/share/emacs/site-lisp/combobulate-interface.el
(defsubst combobulate-node-eq (node1 node2)
  (if (and (combobulate-node-p node1) (combobulate-node-p node2))
      (treesit-node-eq node1 node2)
    (eq node1 node2)))
