(defun embark-completion-list-candidates ()
  "Return all candidates in a completions buffer."
  (embark--with-completion-list-buffer
   (lambda ()
     (cons
      embark--type
      (save-excursion
        (goto-char (point-min))
        (next-completion 1)
        (let (all)
          (while (not (eobp))
            (push (cdr (embark-target-completion-list-candidate)) all)
            (next-completion 1))
          (nreverse all)))))))
