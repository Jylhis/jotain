;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-process.el
(defun majutsu--process-file-insert-filter (marker)
  "Return a process filter that inserts output at MARKER."
  (lambda (_process string)
    (when-let* ((buffer (marker-buffer marker)))
      (with-current-buffer buffer
        (let ((inhibit-read-only t))
          (goto-char marker)
          (insert string)
          (set-marker marker (point)))))))
