Function: combobulate-query-node-text

combobulate-query-node-text is a natively compiled function defined in combobulate-navigation.el.

(combobulate-query-node-text QUERY NODE NODE-ONLY)

;; Defined in /nix/store/9y0jasyzsqjcbfkwqsgdyxcnskjnsypg-emacs-packages-deps/share/emacs/site-lisp/combobulate-navigation.el
(defun combobulate-query-node-text (query node node-only)
  "Executes QUERY against NODE and retrieve their node text

If NODE-ONLY is non-nil then only the node texts are returned"
  (mapcar (lambda (node) (if node-only (combobulate-node-text node)
                      (combobulate-node-text (cdr node))))
          (combobulate-query-search node query t t)))
