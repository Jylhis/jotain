(defun vertico--hilit (cand)
  "Highlight CAND string with lazy highlighting."
  ;; bug#77754: Highlight unquoted string.
  (funcall vertico--hilit (substring (or (get-text-property
                                          0 'completion--unquoted cand) cand))))
