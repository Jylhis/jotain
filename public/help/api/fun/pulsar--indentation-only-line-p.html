;;;; Pulse functions

(defun pulsar--indentation-only-line-p ()
  "Return non-nil if current line has only indentation."
  (let ((inhibit-field-text-motion t))
    (save-excursion
      (goto-char (line-beginning-position))
      (and (not (bobp))
           (or (beginning-of-line 1) t)
           (save-match-data
             (looking-at "^[\s\t]+"))))))
