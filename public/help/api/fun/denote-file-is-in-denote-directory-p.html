;; Defined in /nix/store/5x9606vmybcj80x6qrajp43rma9pa1i7-emacs-packages-deps/share/emacs/site-lisp/elpa/denote-4.2.3/denote.el
(defun denote-file-is-in-denote-directory-p (file)
  "Return non-nil if FILE is in the variable `denote-directory'."
  (seq-some
   (lambda (d)
     (string-prefix-p d (expand-file-name file)))
   (denote-directories)))
