(defun magit-blame--overlay-at (&optional pos key)
  (unless pos
    (setq pos (point)))
  (seq-find (##overlay-get % (or key 'magit-blame-chunk))
            (nconc (overlays-at pos)
                   (overlays-in pos pos))))
