(defun treesit-fold--cons-add (&rest args)
  "Addition for list of cons ARGS."
  (let ((v1 0) (v2 0))
    (dolist (c args)
      (setq v1 (+ v1 (car c))
            v2 (+ v2 (cdr c))))
    (cons v1 v2)))
