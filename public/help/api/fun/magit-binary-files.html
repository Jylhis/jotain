(defun magit-binary-files (&rest args)
  (seq-keep (##and (string-match "^-\t-\t\\(.+\\)" %)
                   (match-str 1 %))
            (apply #'magit-git-items
                   "diff" "-z" "--numstat" "--ignore-submodules"
                   args)))
