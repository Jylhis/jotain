;;; Diagnostic Collection

(defun claude-code-ide-diagnostics--severity-to-vscode (severity)
  "Convert diagnostic SEVERITY to VS Code format.
Returns: 1 (Error), 2 (Warning), 3 (Information), 4 (Hint)."
  (pcase severity
    ;; Flycheck severities
    ('error 1)
    ('warning 2)
    ('info 3)
    ('hint 4)
    ;; Flymake severities
    ('flymake-error 1)
    (':error 1)
    ('flymake-warning 2)
    (':warning 2)
    ('flymake-note 3)
    (':note 3)
    ;; Default
    (_ 3)))
