(defun org-roam-db-insert-tags ()
  "Insert tags for node at point into Org-roam cache."
  (when-let* ((node-id (org-id-get))
              (tags (org-get-tags)))
    (org-roam-db-query [:insert :into tags
                        :values $v1]
                       (mapcar (lambda (tag)
                                 (vector node-id (substring-no-properties tag))) tags))))
