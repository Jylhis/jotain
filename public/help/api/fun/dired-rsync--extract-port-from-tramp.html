(defun dired-rsync--extract-port-from-tramp (file-or-path)
  "Extract the port part of a tramp FILE-OR-PATH."
  (when (tramp-tramp-file-p file-or-path)
    (with-parsed-tramp-file-name file-or-path tfop
      tfop-port)))
