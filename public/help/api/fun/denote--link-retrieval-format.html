(defun denote--link-retrieval-format (file-type)
  "Return link retrieval format based on FILE-TYPE."
  (let ((prop (plist-get
               (alist-get file-type denote-file-types)
               :link-retrieval-format)))
    (if (symbolp prop)
        (symbol-value prop)
      prop)))
