Function: majutsu-jjdescription-show-diff

majutsu-jjdescription-show-diff is an interactive and natively compiled function defined in majutsu-jjdescription.el.

(majutsu-jjdescription-show-diff)

;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-jjdescription.el
(defun majutsu-jjdescription-show-diff ()
  "Show the diff for the described JJ change."
  (interactive)
  (let ((default-directory
         (or (majutsu-jjdescription--bind-repository-root)
             (user-error "No repository associated with this JJ description buffer"))))
    (majutsu-diff-revset (or (majutsu-jjdescription--change-id) "@"))))
