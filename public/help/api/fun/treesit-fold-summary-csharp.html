(defun treesit-fold-summary-csharp (doc-str)
  "Extract summary from DOC-STR in C#."
  (cond ((string-match-p "///" doc-str)
         (treesit-fold-summary-csharp-vsdoc doc-str))
        (t (treesit-fold-summary-javadoc doc-str))))
