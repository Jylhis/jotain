(defun denote--dir-in-denote-directory-p (directory)
  "Return non-nil if DIRECTORY is in variable `denote-directory'."
  (seq-some
   (lambda (d)
     (string-prefix-p d (file-name-as-directory (expand-file-name directory))))
   (denote-directories)))
