;; Defined in /nix/store/mxkjlhf33q1xxjiii3w5v8nn7avms0p9-emacs-packages-deps/share/emacs/site-lisp/elpa/doom-modeline-20260828.1146/doom-modeline-core.el
(defun doom-modeline-string-pixel-width (str)
  "Return the width of STR in pixels."
  (if (fboundp 'string-pixel-width)
      (string-pixel-width str)
    (* (string-width str) (window-font-width nil 'mode-line)
       (if (display-graphic-p) 1.05 1.0))))
