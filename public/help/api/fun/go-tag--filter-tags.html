(defun go-tag--filter-tags (tags options)
  (let ((tag-lst (split-string tags ","))
        (option-alst (mapcar
                      (lambda (opt)
                        (let ((opt-pair (split-string opt "=")))
                          (cons (car opt-pair) (cadr opt-pair))))
                      (split-string options ","))))
    (mapconcat
     'identity
     (delete "" (split-string
                 (mapconcat
                  (lambda (tag)
                    (unless (assoc tag option-alst) tag))
                  tag-lst
                  ",") ","))
     ",")))
