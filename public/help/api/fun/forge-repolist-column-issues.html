(defun forge-repolist-column-issues (spec)
  (and-let* ((repo (forge-get-repository :tracked? nil t))
             (n (caar (forge-sql [:select (funcall count *) :from issue
                                  :where (and (= repository $s1)
                                              (isnull closed))]
                                 (oref repo id)))))
    (magit-repolist-insert-count n spec)))
