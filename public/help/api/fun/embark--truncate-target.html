(defun embark--truncate-target (target)
  "Truncate TARGET string."
  (unless (stringp target)
    (setq target (format "%s" target)))
  (if-let* ((pos (string-match-p "\n" target)))
      (concat (car (split-string target "\n" 'omit-nulls "\\s-*")) "…")
    target))
