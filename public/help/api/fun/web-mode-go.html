(defun web-mode-go (pos &optional offset)
  (unless offset (setq offset 0))
  (when pos
    (cond
      ((and (> offset 0) (<= (+ pos offset) (point-max)))
       (setq pos (+ pos offset)))
      ((and (< offset 0) (>= (+ pos offset) (point-min)))
       (setq pos (+ pos offset)))
      ) ;cond
    (goto-char pos))
  pos)
