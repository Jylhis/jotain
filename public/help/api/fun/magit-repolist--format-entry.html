(defun magit-repolist--format-entry (entry)
  (pcase-let ((`(,id . ,default-directory) entry))
    (list default-directory
          (vconcat
           (mapcar (pcase-lambda (`(,title ,width ,fn ,props))
                     (or (funcall fn `((:id ,id)
                                       (:title ,title)
                                       (:width ,width)
                                       ,@props))
                         ""))
                   magit-repolist-columns)))))
