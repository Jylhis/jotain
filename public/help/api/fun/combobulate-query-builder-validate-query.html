;; Defined in /nix/store/9y0jasyzsqjcbfkwqsgdyxcnskjnsypg-emacs-packages-deps/share/emacs/site-lisp/combobulate-query.el
(cl-defun combobulate-query-builder-validate-query (query)
  "Validate QUERY and return a cons cell if the query is invalid."
  (ignore-errors
    (condition-case err
        (progn (combobulate-query-capture
                (combobulate-root-node)
                query (point-min) (1+ (point-min)))
               nil)
      ;; this can easily happen during startup.
      (treesit-query-error
       (pcase-let ((`(,_ ,message ,start . ,_) err))
         (cons start message))))))
