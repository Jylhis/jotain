(defun tuareg--point-after-comment-p ()
  "Return non-nil if a comment precedes the point."
  (and (eq (char-before) ?\))
       (eq (char-before (1- (point))) ?*) ; implies position is in range
       (save-excursion
         (nth 4 (syntax-ppss (1- (point)))))))
