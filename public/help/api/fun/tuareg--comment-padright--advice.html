(defun tuareg--comment-padright--advice (orig-fun &rest args)
  (let ((str (nth 0 args)))
    (unless (and (derived-mode-p 'tuareg-mode)
                 (stringp str) (not (string-match "\\S-" str)))
      (apply orig-fun args))))
