(defun org-roam--format-nodes-using-function (nodes)
  "Formats NODES using the function org-roam-node-display-template."
  (mapcar (lambda (node)
            (cons
             (propertize (funcall org-roam-node-display-template node) 'node node)
             node))
          nodes)
  )
