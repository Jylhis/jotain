(defun eca-table--count-markup-chars (text)
  "Count hidden bold/italic `*' delimiters in TEXT.
Matches **text** (4 chars) and *text* (2 chars)."
  (let ((count 0)
        (start 0))
    (while (string-match
            "\\(\\*\\{1,2\\}\\)\\([^*].*?\\)\\1"
            text start)
      (setq count (+ count (* 2 (length
                                  (match-string 1 text)))))
      (setq start (match-end 0)))
    count))
