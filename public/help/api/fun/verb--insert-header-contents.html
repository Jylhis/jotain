(defun verb--insert-header-contents (headers)
  "Insert the contents of HTTP HEADERS into the current buffer."
  (let ((inhibit-read-only t))
    (dolist (key-value headers)
      (let ((key (car key-value))
            (value (cdr key-value)))
        (insert key ": " value "\n")))
    (unless (zerop (buffer-size))
      (delete-char -1))))
