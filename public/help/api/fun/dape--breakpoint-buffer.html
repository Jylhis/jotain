;;; Breakpoints

(defun dape--breakpoint-buffer (breakpoint)
  "Return buffer visiting BREAKPOINT if exists."
  (when-let* (((dape--source-breakpoint-p breakpoint))
              (overlay (dape--source-breakpoint-location breakpoint))
              ((overlayp overlay)))
    (overlay-buffer overlay)))
