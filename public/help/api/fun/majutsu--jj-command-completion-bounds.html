;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-command.el
(defun majutsu--jj-command-completion-bounds ()
  "Return minibuffer token bounds for jj command completion."
  (let ((end (point)))
    (save-excursion
      (skip-chars-backward "^   \n" (minibuffer-prompt-end))
      (cons (point) end))))
