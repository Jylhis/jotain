(defun eca-workspaces--restore-point (entity column)
  "Move point to ENTITY line at COLUMN, falling back to buffer start."
  (if-let* ((position (eca-workspaces--find-entity entity)))
      (progn
        (goto-char position)
        (move-to-column (or column 0)))
    (goto-char (point-min))))
