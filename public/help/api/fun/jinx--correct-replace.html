(defun jinx--correct-replace (overlay word)
  "Replace OVERLAY with WORD."
  (when-let* ((start (overlay-start overlay))
              (end (overlay-end overlay)))
    (undo-boundary)
    (delete-overlay overlay)
    (goto-char end)
    (insert-before-markers word)
    (delete-region start end)))
