(majutsu-template-defun if ((condition Boolean)
                            (then Any)
                            (else Any :optional t))
  (:returns Any :doc "if(COND, THEN [, ELSE]).")
  (majutsu-template--call-node
   "if"
   (if else
       (list condition then else)
     (list condition then))
   'Any
   '(:conditional t)))
