Function: pdf-view-active-region-text

pdf-view-active-region-text is a natively compiled function defined in pdf-view.el.

;; Defined in /nix/store/wm0vc3dfi6c7pwjhcfpsnhyg4hd0b5yc-emacs-packages-deps/share/emacs/site-lisp/elpa/pdf-tools-20260102.1101/pdf-view.el
(defun pdf-view-active-region-text ()
  "Return the text of the active region as a list of strings."
  (pdf-view-assert-active-region)
  (mapcar
   (lambda (edges)
     (pdf-info-gettext
      (car pdf-view-active-region)
      edges
      pdf-view-selection-style))
   (cdr pdf-view-active-region)))
