(defun magit-todos--buffer-items (results-regexp)
  "Return list of `magit-todos-item' found in current buffer for RESULTS-REGEXP."
  (let ((items))
    (save-excursion
      (goto-char (point-min))
      (while (not (eobp))
        (--when-let (magit-todos--line-item results-regexp)
          (push it items))
        (forward-line 1)))
    (nreverse items)))
