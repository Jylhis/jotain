(defun majutsu-jjdescription--stat-line-re ()
  "Return a regexp matching JJ diffstat summary lines."
  (concat (majutsu-jjdescription--comment-prefix-re)
          "\\([^|]+\\)\\s-*|\\s-*\\(.+\\)$"))
