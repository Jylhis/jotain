;; Entry point

(defun eca-settings--find-window ()
  "Find a window displaying a settings buffer, or nil."
  (let ((found nil))
    (walk-windows
     (lambda (win)
       (when (buffer-local-value 'eca-settings--tab-key
                                 (window-buffer win))
         (setq found win)))
     nil t)
    found))
