(defun eca-rewrite--time->presentable-time (ms)
  "Return a presentable time string for MS."
  (let ((secs (/ ms 1000)))
    (if (< secs 60)
        (format "%ds" secs)
      (let ((mins (/ secs 60))
            (rem  (% secs 60)))
        (if (zerop rem)
            (format "%dm" mins)
          (format "%dm %ds" mins rem))))))
