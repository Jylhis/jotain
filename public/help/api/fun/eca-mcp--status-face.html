(defun eca-mcp--status-face (status)
  "Return the face used to display MCP STATUS."
  (pcase status
    ("running" 'success)
    ("starting" 'warning)
    ("failed" 'error)
    ((or "stopped" "stopping" "disabled") 'shadow)
    ("requires-auth" 'eca-mcp-details-requires-auth-face)
    (_ 'shadow)))
