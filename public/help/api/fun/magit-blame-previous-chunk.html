(defun magit-blame-previous-chunk ()
  "Move to the previous chunk."
  (interactive)
  (if-let ((prev (previous-single-char-property-change
                  (point) 'magit-blame-chunk)))
      (goto-char prev)
    (user-error "No more chunks")))
