(defun magit-run-hook-with-benchmark (hook)
  (cond
    ((not hook))
    (magit-refresh-verbose
     (message "Running %s..." hook)
     (message "Running %s...done (%.3fs)" hook
              (benchmark-elapse
                (run-hook-wrapped
                 hook
                 (lambda (fn)
                   (message "  %-50s %f" fn (benchmark-elapse (funcall fn))))))))
    ((run-hooks hook))))
