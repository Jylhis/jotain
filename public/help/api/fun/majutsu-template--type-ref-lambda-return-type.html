;; Defined in /nix/store/bz86lcx20lhz4l1xw4qxxfrpzz1ibl7c-emacs-packages-deps/share/emacs/site-lisp/majutsu-template.el
(defun majutsu-template--type-ref-lambda-return-type (type)
  "Return lambda return type carried by TYPE, or nil."
  (setq type (majutsu-template--type-ref-normalize type))
  (and (consp type)
       (eq (car type) :lambda)
       (caddr type)))
