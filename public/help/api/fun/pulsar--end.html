(defun pulsar--end ()
  "Return appropriate line end."
  (let ((inhibit-field-text-motion t))
    (if (and (pulsar--buffer-end-p) (eq (char-before) ?\n))
        (line-beginning-position 1)
      (line-beginning-position 2))))
