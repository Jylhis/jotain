(defun gnuplot--close-down ()
  "Tidy up when deleting the gnuplot buffer."
  (if (and gnuplot-process
           (eq (process-status gnuplot-process) 'run))
      (kill-process gnuplot-process))
  (setq gnuplot-process nil
        gnuplot-buffer nil))
