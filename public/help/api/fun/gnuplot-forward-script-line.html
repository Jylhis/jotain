(defun gnuplot-forward-script-line (&optional num)
  "Move forward my NUM script lines.
Blank lines and commented lines are not included in the NUM count."
  (interactive "p" gnuplot-mode)
  (while (> num 0)
    (and (not (eobp)) (forward-line 1))
    (while (and (not (eobp))
                (or (looking-at "^\\s-*$")
                    (looking-at "^\\s-*#")))
      (forward-line 1))
    (setq num (1- num))))
