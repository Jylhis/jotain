(defun majutsu-template--call-with-bindings (bindings thunk)
  "Call THUNK with lexical BINDINGS temporarily active."
  (let ((majutsu-template--binding-stack
         (append bindings majutsu-template--binding-stack)))
    (funcall thunk)))
