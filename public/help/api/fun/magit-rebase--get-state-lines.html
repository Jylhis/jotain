(defun magit-rebase--get-state-lines (file)
  (and (magit-rebase-in-progress-p)
       (let ((dir (magit-gitdir)))
         (magit-file-line
          (expand-file-name
           (concat (if (file-directory-p (expand-file-name "rebase-merge" dir))
                       "rebase-merge/"
                     "rebase-apply/")
                   file)
           dir)))))
