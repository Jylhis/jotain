;;;; Predicates

(defun jinx--regexp-excluded-p (pos)
  "Return non-nil if word at POS matches a regexp from `jinx-exclude-regexps'."
  (save-excursion
    (goto-char pos)
    (when (and jinx--exclude-regexp (looking-at-p jinx--exclude-regexp))
      (save-match-data
        (looking-at jinx--exclude-regexp)
        (match-end 0)))))
