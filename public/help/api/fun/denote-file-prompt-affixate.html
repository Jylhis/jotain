(defun denote-file-prompt-affixate (files)
  "Affixate FILES.
Use the identifier as a prefix and the keywords as a suffix."
  (mapcar
   (lambda (file)
     (list
      file
      (or (denote-file-prompt--format-identifier file) "")
      (or (denote-file-prompt--format-keywords-and-signature file) "")))
   files))
