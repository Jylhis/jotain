Function: web-mode-is-css-string

web-mode-is-css-string is a natively compiled function defined in web-mode.el.

(web-mode-is-css-string POS)

;; Defined in /nix/store/x925hd4kr65vw6221bf7bb2azagpjif5-emacs-packages-deps/share/emacs/site-lisp/elpa/web-mode-20260831.219/web-mode.el
(defun web-mode-is-css-string (pos)
  (let (beg)
    (cond
      ((and (setq beg (web-mode-part-token-beginning-position pos))
            (web-mode-looking-at-p "`" beg)
            (web-mode-looking-back "\\(styled[[:alnum:].]+\\|css\\)" beg))
       beg)
      (t
       nil)
      ) ;cond
    ))
