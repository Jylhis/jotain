(defun cape-wrap-inside-faces (capf &rest faces)
  "Call CAPF only if inside FACES."
  (when-let* (((> (point) (point-min)))
              (fs (get-text-property (1- (point)) 'face))
              ((if (listp fs)
                   (cl-loop for f in fs thereis (memq f faces))
                 (memq fs faces))))
    (funcall capf)))
