(defun pdf-util-temp-prefix ()
  "Create a temp-file prefix for the current buffer"
  (concat (if buffer-file-name
              (file-name-nondirectory buffer-file-name)
            (replace-regexp-in-string "[^[:alnum:]]+" "-" (buffer-name)))
          "-"))
