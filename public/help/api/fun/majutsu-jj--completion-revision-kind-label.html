(defun majutsu-jj--completion-revision-kind-label (kind)
  "Return display label for revision completion KIND."
  (pcase kind
    ('bookmark "bookmark")
    ('remote-bookmark "remote bookmark")
    ('tag "tag")
    ('workspace "workspace")
    ('change-id "change")
    ('revset-alias "alias")
    ((pred stringp) kind)
    (_ nil)))
