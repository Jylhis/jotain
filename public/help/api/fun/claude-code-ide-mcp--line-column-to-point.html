Convert LINE and COLUMN to point position in current buffer.

LINE is 1-based, COLUMN is 0-based (Emacs convention).

(defun claude-code-ide-mcp--line-column-to-point (line column)
  "Convert LINE and COLUMN to point position in current buffer.
LINE is 1-based, COLUMN is 0-based (Emacs convention)."
  (save-excursion
    (goto-char (point-min))
    (forward-line (1- line))
    (move-to-column column)
    (point)))
