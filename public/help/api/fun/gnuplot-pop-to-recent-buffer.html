(defun gnuplot-pop-to-recent-buffer ()
  "Switch to the most recently-plotted gnuplot script buffer."
  (interactive)
  (when (buffer-live-p gnuplot--comint-recent-buffer)
    (pop-to-buffer gnuplot--comint-recent-buffer)))
