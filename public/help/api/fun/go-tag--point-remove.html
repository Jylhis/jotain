(defun go-tag--point-remove (point tags &optional options)
  "Add field TAGS for the struct under the POINT."
  (let ((cmd-args (append
                   go-tag-args
                   (list "-offset" (format "%S" point)))))
    (go-tag--remove cmd-args tags options)))
