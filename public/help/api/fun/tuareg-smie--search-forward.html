(defun tuareg-smie--search-forward (tokens)
  (let (tok)
    (while (progn
             (setq tok (tuareg-smie--forward-token))
             (if (not (zerop (length tok)))
                 (not (member tok tokens))
               (unless (eobp)
                 (condition-case err
                     (progn (forward-sexp) t)
                   (scan-error
                    (setq tok (buffer-substring (nth 2 err) (nth 3 err)))
                    nil))))))
    tok))
