(defun magit-read-files (prompt initial-input history &optional list-fn)
  (magit-completing-read-multiple prompt
                                  (funcall (or list-fn #'magit-list-files))
                                  nil nil
                                  (or initial-input (magit-file-at-point))
                                  history))
