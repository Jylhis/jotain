(defun projection-autotools-run-configure ()
  "Configure command generator for autotools projects."
  (projection--join-shell-commands
   `(,(if (file-exists-p "autogen.sh")
          "./autogen.sh"
        "autoconf")
     ("./configure" ,@projection-autotools-configure-options))))
