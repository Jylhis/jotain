(defun ansible-list-playbooks ()
  "Find .yml files in `ansible-root-path`."
  (if (ansible-update-root-path)
      (mapcar
       (lambda (file) (f-relative file ansible-root-path))
       (f-files ansible-root-path (lambda (file)
                                    (s-matches? ".yml" (f-long file)))
                t))
    nil))
