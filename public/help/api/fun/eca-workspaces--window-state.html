(defun eca-workspaces--window-state (window)
  "Return (WINDOW ENTITY COLUMN) for WINDOW's current point."
  (let ((position (window-point window)))
    (list window
          (eca-workspaces--entity-at position)
          (save-excursion
            (goto-char position)
            (current-column)))))
