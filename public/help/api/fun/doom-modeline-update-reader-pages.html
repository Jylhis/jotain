(defun doom-modeline-update-reader-pages ()
  "Update Emacs Reader document pages."
  (setq doom-modeline--reader-pages
        (format "  P%d/P%d "
                (or (eval `(reader-current-doc-pagenumber)) 0)
                (or reader-current-doc-pagecount 0))))
