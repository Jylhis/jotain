(defun eca-rewrite-accept ()
  "Accept rewrite overlay at point."
  (interactive)
  (if-let ((ov (eca-rewrite--overlay-at-point)))
    (eca-rewrite--accept ov)
    (eca-error "No rewrite overlay found at point")))
