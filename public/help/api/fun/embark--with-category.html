;; Defined in /nix/store/fs4fhphi5ajjmvzx1rm550ibr8slmvxd-emacs-packages-deps/share/emacs/site-lisp/elpa/embark-20260610.302/embark.el
(defun embark--with-category (category candidates)
  "Return completion table for CANDIDATES of CATEGORY with sorting disabled."
  (completion-table-with-metadata
   candidates
   `((display-sort-function . identity)
     (cycle-sort-function . identity)
     (category . ,category))))
