(defun expreg--keyboard-quit-advice ()
  "Restores point when ‘keyboard-quit’ is called."
  (interactive)
  (when (and expreg-restore-point-on-quit expreg--initial-point)
    (goto-char expreg--initial-point))
  (setq expreg--initial-point nil))
