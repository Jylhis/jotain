;; Defined in /nix/store/1jkgkg7y6y6hp0xia3dsmcl5zaj0bnhq-emacs-packages-deps/share/emacs/site-lisp/ghostel.el
(defun ghostel--window-buffer-pair-live-p (entry)
  "Return non-nil if ENTRY's window still shows ENTRY's buffer."
  (let ((window (car entry))
        (buffer (cdr entry)))
    (and (window-live-p window)
         (eq (window-buffer window) buffer))))
