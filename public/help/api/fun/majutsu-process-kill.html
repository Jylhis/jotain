(defun majutsu-process-kill ()
  "Kill the process at point."
  (interactive)
  (when-let* ((process (magit-section-value-if 'process)))
    (unless (eq (process-status process) 'run)
      (user-error "Process isn't running"))
    (kill-process process)))
