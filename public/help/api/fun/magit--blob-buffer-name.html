(defun magit--blob-buffer-name (obj file &optional volatile relative)
  (format "%s%s.~%s~"
          (if volatile " " "")
          (or (if relative (file-relative-name file (magit-toplevel)) file)
              (and (magit-blob-p obj) "{blob}"))
          (subst-char-in-string ?/ ?_ obj)))
