(cl-defmethod forge--fetch-issue-posts ((repo forge-gitlab-repository) cur cb)
  (forge--glab-get repo
    (let-alist (car cur)
      (format "/projects/%s/issues/%s/notes" .project_id .iid))
    '((per_page . 100))
    :unpaginate t
    :callback (lambda (value)
                (setf (alist-get 'notes (car cur)) value)
                (funcall cb))))
