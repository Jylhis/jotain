(defun eca-settings--refresh-current ()
  "Refresh the currently displayed settings tab."
  (interactive)
  (when-let* ((session (ignore-errors (eca-session)))
              (key eca-settings--tab-key))
    (eca-settings-refresh-tab key session)))
