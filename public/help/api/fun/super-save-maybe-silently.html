Function: super-save-maybe-silently

(super-save-maybe-silently FN)

(defun super-save-maybe-silently (fn)
  "Call FN, suppressing messages if `super-save-silent' is non-nil."
  (if super-save-silent
      (with-temp-message ""
        (let ((inhibit-message t)
              (inhibit-redisplay t)
              (message-log-max nil))
          (funcall fn)))
    (funcall fn)))
