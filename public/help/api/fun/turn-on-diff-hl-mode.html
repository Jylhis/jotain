(defun turn-on-diff-hl-mode ()
  "Turn on `diff-hl-mode' or `diff-hl-dir-mode' in a buffer if appropriate."
  (let ((file (diff-hl--buffer-file-name)))
    (cond
     (file
      (unless (and diff-hl-disable-on-remote
                   (file-remote-p file))
        (diff-hl-mode 1)))
     ((eq major-mode 'vc-dir-mode)
      (diff-hl-dir-mode 1)))))
