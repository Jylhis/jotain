(defun tuareg--skip-forward-comments-semicolon ()
  "Skip `;;' and then `sticky' comments after a definition."
  (when (looking-at (rx (* (in " \t\n")) ";;"))
    (goto-char (match-end 0)))
  (while (tuareg--skip-forward-comment)))
