(defun logview--help-substitute-keys (text)
  (replace-regexp-in-string (rx "\\["
                                (group (1+ (any alnum ?-)))
                                (? " <" (group (1+ (not (any ?>)))) ">")
                                "]")
                            (lambda (text)
                              (save-match-data
                                (logview--help-format-keys (list (intern (match-string 1 text))) (match-string 2 text))))
                            text t t))
