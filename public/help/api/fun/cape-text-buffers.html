(defun cape-text-buffers ()
  "Return `text-mode' and `prog-mode' buffers."
  (cape--buffer-list
   (lambda (buf)
     (let ((mode (buffer-local-value 'major-mode buf)))
       (or (provided-mode-derived-p mode #'text-mode)
           (provided-mode-derived-p mode #'prog-mode))))))
