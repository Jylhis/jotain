(defun markdown-metadata-line-p (pos regexp)
  (save-excursion
    (or (= (line-number-at-pos pos) 1)
        (progn
          (forward-line -1)
          ;; skip multi-line metadata
          (while (and (looking-at-p "^\\s-+[[:alpha:]]")
                      (> (line-number-at-pos (point)) 1))
            (forward-line -1))
          (looking-at-p regexp)))))
