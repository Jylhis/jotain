(defun magit--client-message (format-string &rest args)
  ;; See `server-process-filter'.
  (let ((msg (format "-print %s\n"
                     (server-quote-arg
                      (apply #'format-message format-string args)))))
    (dolist (client server-clients)
      (server-send-string client msg))))
