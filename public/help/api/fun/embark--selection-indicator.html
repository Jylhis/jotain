(defun embark--selection-indicator ()
  "Mode line indicator showing number of selected items."
  (when-let* ((sel
               (buffer-local-value
                'embark--selection
                (or (when-let* ((win (active-minibuffer-window)))
                      (window-buffer win))
                    (current-buffer)))))
    (format embark-selection-indicator (length sel))))
