(defun magit--find-buffer (&rest plist)
  "Like `find-buffer' but take multiple VARIABLE-VALUE pairs."
  (seq-find (lambda (buf)
              (let ((plist plist))
                (while (and plist
                            (equal (buffer-local-value (car plist) buf)
                                   (cadr plist)))
                  (setq plist (cddr plist)))
                (not plist)))
            (buffer-list)))
