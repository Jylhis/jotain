(defun org-roam-strip-comments (s)
  "Strip Org comments from string S."
  (with-temp-buffer
    (insert s)
    (goto-char (point-min))
    (while (not (eobp))
      (if (org-at-comment-p)
          (delete-region (line-beginning-position)
                         (progn (forward-line) (point)))
        (forward-line)))
    (buffer-string)))
