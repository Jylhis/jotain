(defun helpful--variable-at-point ()
  "Return the variable exactly under point, or defined at point."
  (let ((var (helpful--variable-at-point-exactly)))
    (if var
        var
      (let ((var (helpful--variable-defined-at-point)))
        (when (helpful--variable-p var)
          var)))))
