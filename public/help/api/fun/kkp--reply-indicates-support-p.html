(defun kkp--reply-indicates-support-p (reply)
  "Return non-nil when REPLY is a well-formed KKP `CSI ? flags u' response."
  (and (member (length reply) '(5 6))
       (equal '(27 91 63) (cl-subseq reply 0 3))
       (eql 117 (car (last reply)))))
