(defun majutsu-jjdescription-setup-font-lock ()
  "Set up font-lock for JJ description buffers."
  (if majutsu-jjdescription-mode
      (majutsu-jjdescription--refresh-font-lock)
    (majutsu-jjdescription-mode 1)))
