(defun majutsu-diff--goto-line-col (buffer line col)
  "Move point in BUFFER to LINE and COL."
  (with-current-buffer buffer
    (widen)
    (goto-char (point-min))
    (forward-line (max 0 (1- line)))
    (move-to-column col)))
