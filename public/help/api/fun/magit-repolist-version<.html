(defun magit-repolist-version< (a b)
  (save-match-data
    (let ((re "[0-9]+\\(\\.[0-9]*\\)*"))
      (setq a (and (string-match re a) (match-str 0 a)))
      (setq b (and (string-match re b) (match-str 0 b)))
      (cond ((and a b) (version< a b))
            (b nil)
            (t)))))
