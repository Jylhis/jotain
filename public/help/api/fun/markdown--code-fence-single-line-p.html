(defun markdown--code-fence-single-line-p (begin)
  (save-excursion
    (goto-char begin)
    (save-match-data
      (and (re-search-forward "`\\{3,\\}" nil t)
           (re-search-forward "`\\{3,\\}" (line-end-position) t)))))
