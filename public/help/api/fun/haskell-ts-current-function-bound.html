(defun haskell-ts-current-function-bound ()
  "Get start and end point of current funciton."
  (let (start end)
    (save-excursion
      (mark-defun)
      (setq start (region-beginning))
      (setq end (region-end))
      (deactivate-mark))
    (list start end)))
