(defun magit-read-remote-or-url (prompt &optional default)
  (magit-completing-read prompt
                         (nconc (magit-list-remotes)
                                (list "https://" "git://" "git@"))
                         nil 'any nil nil
                         (or default
                             (magit-remote-at-point)
                             (magit-get-remote))))
