(defun mc/cursor-is-bar ()
  "Return non-nil if the cursor is a bar."
  (let ((cursor-type
         (if (eq cursor-type t)
             (frame-parameter nil 'cursor-type)
           cursor-type)))
    (or (eq cursor-type 'bar)
        (and (listp cursor-type)
             (eq (car cursor-type) 'bar)))))
