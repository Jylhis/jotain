(defun majutsu-template--method-name-from-node (node)
  "Return normalized method name string from NODE."
  (let ((raw (majutsu-template--node->identifier node "method name")))
    (if (and (> (length raw) 0) (eq (aref raw 0) ?:))
        (substring raw 1)
      raw)))
