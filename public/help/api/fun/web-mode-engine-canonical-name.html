(defun web-mode-engine-canonical-name (name)
  (let (engine)
    (cond
      ((null name)
       nil)
      ((assoc name web-mode-engines)
       name)
      (t
       (dolist (elt web-mode-engines)
         (when (and (null engine) (member name (cdr elt)))
           (setq engine (car elt)))
         ) ;dolist
       engine)
      )))
