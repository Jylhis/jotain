(defun eca-rewrite--normalize-start-region (start)
  "Normalize START region.
If chars from START until bol are all spaces return bol,
otherwise START."
  (save-excursion
    (goto-char start)
    (let ((bol (line-beginning-position)))
      (if (string-match-p "\\`[ \t]*\\'"
                          (buffer-substring bol start))
          bol
        start))))
