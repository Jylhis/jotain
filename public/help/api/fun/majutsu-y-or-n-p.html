(defun majutsu-y-or-n-p (prompt &optional action)
  "Ask PROMPT with y/n or yes/no depending on ACTION settings."
  (if (or (eq majutsu-slow-confirm t)
          (and action (member action majutsu-slow-confirm)))
      (yes-or-no-p prompt)
    (y-or-n-p prompt)))
