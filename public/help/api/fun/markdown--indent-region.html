(defun markdown--indent-region (start end)
  (let ((deactivate-mark nil))
    (save-excursion
      (goto-char end)
      (setq end (point-marker))
      (goto-char start)
      (when (bolp)
        (forward-line 1))
      (while (< (point) end)
        (unless (or (markdown-code-block-at-point-p) (and (bolp) (eolp)))
          (indent-according-to-mode))
        (forward-line 1))
      (move-marker end nil))))
