(defun eca-chat--collapse-question-block (text)
  "Replace the standalone question block with TEXT."
  (when-let* ((ov (eca-chat--question-block-ov)))
    (let ((start (overlay-start ov))
          (end (overlay-end ov))
          (inhibit-read-only t))
      (delete-overlay ov)
      (save-excursion
        (goto-char start)
        (delete-region start end)
        (eca-chat--insert text)))))
