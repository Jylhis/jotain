Function: claude-code-ide--set-window-height

claude-code-ide--set-window-height is an interactive and natively compiled function defined in claude-code-ide-transient.el.

;; Defined in /nix/store/ca4a4maamih3h4ksbbhy6racb61djy6q-emacs-packages-deps/share/emacs/site-lisp/claude-code-ide-transient.el
(transient-define-suffix claude-code-ide--set-window-height (height)
  "Set window height."
  :description "Set window height"
  (interactive (list (read-number "Window height: " claude-code-ide-window-height)))
  (setq claude-code-ide-window-height height)
  (claude-code-ide-log "Window height set to %d" height))
