(defun magit--abbrev-if-oid (obj)
  (cond ((member obj '("{index}" "{worktree}")) obj)
        ((magit-ref-p obj) obj)
        ((magit-rev-parse (magit-abbrev-arg "short") obj))
        (obj)))
