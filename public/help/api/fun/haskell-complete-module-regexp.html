(defun haskell-complete-module-regexp (pattern)
  "Make a regular expression for the given module pattern. Example:

\"c.m.s\" -> \"^c[^.]*\\.m[^.]*\\.s[^.]*\"

"
  (let ((components (mapcar #'haskell-complete-module-component
                            (split-string pattern "\\." t))))
    (concat "^"
            (mapconcat #'identity
                       components
                       "\\."))))
