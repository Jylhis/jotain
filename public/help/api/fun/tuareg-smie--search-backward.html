(defun tuareg-smie--search-backward (tokens)
  (let (tok)
    (while (progn
             (setq tok (tuareg-smie--backward-token))
             (if (not (zerop (length tok)))
                 (not (member tok tokens))
               (unless (bobp)
                 (condition-case err
                     (progn (backward-sexp) t)
                   (scan-error
                    (setq tok (buffer-substring (nth 3 err) (1+ (nth 3 err))))
                    nil))))))
    tok))
