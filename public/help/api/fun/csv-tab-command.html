(defun csv-tab-command ()
  "Skip to the next field on the same line.
Create a new field at end of line, if needed."
  (interactive)
  (skip-chars-forward csv--skip-chars)
  (if (eolp)
      (insert (car csv-separators))
    (forward-char 1)))
