(defun eca-rewrite-merge ()
  "Merge between original text and rewrite overlay OV."
  (interactive)
  (if-let ((ov (eca-rewrite--overlay-at-point)))
    (eca-rewrite--merge ov)
    (eca-error "No rewrite overlay found at point")))
