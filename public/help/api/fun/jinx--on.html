(defun jinx--on ()
  "Turn `jinx-mode' on."
  (unless (or noninteractive
              buffer-read-only
              (buffer-base-buffer) ;; Do not enable in indirect buffers
              (eq (aref (buffer-name) 0) ?\s))
    (jinx-mode)))
