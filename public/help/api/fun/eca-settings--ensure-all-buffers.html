(defun eca-settings--ensure-all-buffers (session)
  "Ensure all registered tab buffers exist for SESSION."
  (dolist (tab eca-settings--tabs)
    (let ((key (plist-get tab :key)))
      (eca-settings--get-or-create-tab-buffer key session))))
