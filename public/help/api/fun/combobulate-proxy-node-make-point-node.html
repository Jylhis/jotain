(defun combobulate-proxy-node-make-point-node (&optional pt)
  "Create a proxy node at `point'."
  (combobulate-proxy-node-create
   :start (or pt (point))
   :end (or pt (point))
   :text ""
   :type "point"
   :named t
   :node nil
   :field nil
   :pp "Point"))
