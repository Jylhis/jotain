(defun claude-code-ide-mcp--active-sessions ()
  "Return a list of all active MCP sessions."
  (let ((sessions '()))
    (maphash (lambda (_project-dir session)
               (push session sessions))
             claude-code-ide-mcp--sessions)
    sessions))
