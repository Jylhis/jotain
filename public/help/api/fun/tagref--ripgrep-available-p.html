(defun tagref--ripgrep-available-p ()
  "Return non-nil if ripgrep (rg) is available."
  (when (eq tagref--ripgrep-available 'unknown)
    (setq tagref--ripgrep-available (executable-find "rg")))
  tagref--ripgrep-available)
