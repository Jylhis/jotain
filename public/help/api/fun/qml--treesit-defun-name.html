Function: qml--treesit-defun-name

qml--treesit-defun-name is a natively compiled function defined in qml-ts-mode.el.

Signature

(qml--treesit-defun-name NODE)

Documentation

Return the defun name of NODE.

Return nil if there is no name or if NODE is not a defun node.

;; Defined in /nix/store/9jq2m0hljqa2mm6l3v2h9k2h59hvj0nc-emacs-packages-deps/share/emacs/site-lisp/qml-ts-mode.el
(defun qml--treesit-defun-name (node)
  "Return the defun name of NODE.
Return nil if there is no name or if NODE is not a defun node."
  (treesit-node-text
   (pcase (treesit-node-type node)
     ("lexical_declaration"
      (treesit-node-child-by-field-name
       (treesit-search-subtree node "variable_declarator" nil nil 1) "name"))
     ((or "function_declaration" "method_definition" "class_declaration" "ui_signal" "ui_property")
      (treesit-node-child-by-field-name node "name"))
     ("ui_object_definition"
      (treesit-node-child-by-field-name node "type_name")))
   t))
