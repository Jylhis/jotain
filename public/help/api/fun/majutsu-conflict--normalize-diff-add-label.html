(defun majutsu-conflict--normalize-diff-add-label (label)
  "Strip JJ diff prefix from add LABEL."
  (let ((label (and label (replace-regexp-in-string "\\`[ \t]+" "" label))))
    (if (and label (string-match "\\`to:[ \t]*\\(.*\\)\\'" label))
        (match-string 1 label)
      label)))
