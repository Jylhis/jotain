(defun eca-chat-inline--fallback-command ()
  "Return the pressed key's binding outside the viewport mode."
  (let ((keys (this-command-keys-vector)))
    (when (> (length keys) 0)
      (let ((eca-chat-inline--viewport-mode nil))
        (key-binding keys t)))))
