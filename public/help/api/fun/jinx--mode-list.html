(defun jinx--mode-list (list)
  "Lookup by mode (major or minor) in LIST."
  (cl-loop for (mode . vals) in list
           if (or (and (boundp mode) (symbol-value mode)) (derived-mode-p mode))
           append (if (symbolp vals) (alist-get vals list) vals)))
