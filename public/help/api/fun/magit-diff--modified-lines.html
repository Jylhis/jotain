(defun magit-diff--modified-lines (bound)
  (let (-lines +lines)
    (while (re-search-forward
            "^@@ -\\([0-9]+\\),[^ ]+ \\+\\([0-9]+\\),[^ ]+ @@.*\n" bound t)
      (let ((-line (string-to-number (match-string 1)))
            (+line (string-to-number (match-string 2))))
        (while (and (not (eobp))
                    (memq (char-after) '(?\s ?- ?+)))
          (pcase (char-after)
            (?\s (incf -line)
                 (incf +line))
            (?-  (push (incf -line) -lines))
            (?+  (push (incf +line) +lines)))
          (forward-line))))
    (list (nreverse -lines)
          (nreverse +lines))))
