combobulate-query-pretty-print is a natively compiled function defined in combobulate-query.el.

(combobulate-query-pretty-print QUERY &optional PARSER)

Source Code

;; Defined in /nix/store/9y0jasyzsqjcbfkwqsgdyxcnskjnsypg-emacs-packages-deps/share/emacs/site-lisp/combobulate-query.el
(defun combobulate-query-pretty-print (query &optional parser)
  "Pretty print QUERY using the given PARSER, or the first parser if none is given."
  (combobulate-query-builder-fontify-query
   (combobulate-query-builder-to-string query)
   (or parser (car (combobulate-parser-list)))))
