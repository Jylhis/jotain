;; Defined in /nix/store/mxkjlhf33q1xxjiii3w5v8nn7avms0p9-emacs-packages-deps/share/emacs/site-lisp/elpa/doom-modeline-20260828.1146/doom-modeline-segments.el
(defun doom-modeline--unicode-number (num)
  "Return a nice unicode representation of NUM."
  (cond
   ((not num) "")
   ((string= "1" num) "➊")
   ((string= "2" num) "➋")
   ((string= "3" num) "➌")
   ((string= "4" num) "➍")
   ((string= "5" num) "➎")
   ((string= "6" num) "➏")
   ((string= "7" num) "➐")
   ((string= "8" num) "➑")
   ((string= "9" num) "➒")
   ((string= "10" num) "➓")
   (t num)))
