(defun eca-mcp--server-state-label (server)
  "Return the operational state label for MCP SERVER."
  (pcase (plist-get server :status)
    ("running" "Running")
    ("starting" "Starting")
    ("stopping" "Stopping")
    ((or "stopped" "disabled") "Stopped")
    ("failed" "Failed")
    ("requires-auth" "Needs auth")
    (_ "Unknown")))
