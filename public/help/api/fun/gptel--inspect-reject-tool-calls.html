(defun gptel--inspect-reject-tool-calls (&optional _)
  "Cancel tool-calls and return to query buffer."
  (interactive)
  (apply #'gptel--reject-tool-calls
   (thread-first (gptel-fsm-info gptel--fsm-last)
                 (plist-get :tool-display)))
  (quit-window t))
