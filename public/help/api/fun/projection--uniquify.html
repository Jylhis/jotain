(defun projection--uniquify (collection)
  "Helper to remove duplicates from COLLECTION."
  (thread-last
    collection
    (append (list (make-hash-table :test 'equal)))
    (cl-reduce (lambda (hash-table target)
                 (puthash target t hash-table)
                 hash-table))
    (hash-table-keys)))
