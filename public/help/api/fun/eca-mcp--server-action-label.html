(defun eca-mcp--server-action-label (server)
  "Return the user-facing runtime action label for MCP SERVER."
  (pcase (plist-get server :status)
    ("failed" "Retry")
    ("stopping" "Wait")
    (_ (or (pcase (eca-mcp--server-runtime-action server)
             ('start "Start")
             ('stop "Stop")
             ('connect "Connect"))
           "Unavailable"))))
