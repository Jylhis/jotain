(defun markdown--dnd-multi-local-file-handler (urls action)
  (let ((multile-urls-p (> (length urls) 1)))
    (dolist (url urls)
      (markdown--dnd-local-file-handler url action)
      (when multile-urls-p
        (insert " ")))))
