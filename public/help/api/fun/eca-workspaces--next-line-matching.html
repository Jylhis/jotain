(defun eca-workspaces--next-line-matching (predicate)
  "Move to the next line matching PREDICATE, or stay put."
  (let ((origin (point))
        (found nil))
    (forward-line 1)
    (while (and (not (eobp)) (not found))
      (if (funcall predicate)
          (setq found t)
        (forward-line 1)))
    (unless found
      (goto-char origin))
    found))
