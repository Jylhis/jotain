(defun eca-rewrite-reject ()
  "Reject rewrite overlay at point."
  (interactive)
  (if-let ((ov (eca-rewrite--overlay-at-point)))
    (eca-rewrite--reject (list ov))
    (eca-error "No rewrite overlay found at point")))
