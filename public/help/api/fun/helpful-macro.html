;;;###autoload
(defun helpful-macro (symbol)
  "Show help for macro named SYMBOL."
  (interactive
   (list (helpful--read-symbol
          "Macro: "
          (helpful--callable-at-point)
          #'macrop)))
  (helpful--update-and-switch-buffer symbol t))
