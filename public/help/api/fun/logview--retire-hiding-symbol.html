(defun logview--retire-hiding-symbol (symbol-var)
  (set symbol-var (intern (replace-regexp-in-string "[0-9]+$" (lambda (generation) (number-to-string (1+ (string-to-number generation))))
                                                    (symbol-name (symbol-value symbol-var)) t t))))
