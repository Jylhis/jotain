(defun eca-doom--status-face (status)
  "Return the face to apply for STATUS, or nil when idle."
  (pcase status
    ('waiting-approval 'eca-doom-workspace-tab-attention-face)
    ('running 'eca-doom-workspace-tab-running-face)
    (_ nil)))
