Narrowing configuration for compile-multi.

This configuration lets you filter down completion candidates exposed by compile-multi. Each entry should be of the form (char name types...).

;; Defined in /nix/store/jr3v0pan2wjm5qwpkdv1rghw5aw9k6i9-emacs-packages-deps/share/emacs/site-lisp/elpa/consult-compile-multi-20250101.2156/consult-compile-multi.el
(defcustom consult-compile-multi-narrow
  '((?p "Project" project)
    (?P "Pytest" pytest-file pytest-class pytest-test)
    (?c "CMake" cmake ctest)
    (?e "Emacs" emacs)
    (?t "Test" ctest))
  "Narrowing configuration for `compile-multi'.
This configuration lets you filter down completion candidates exposed by
`compile-multi'. Each entry should be of the form (char name types...)."
  :type '(alist
          :key-type (character :tag "Character to narrow on")
          :value-type
          (cons (string :tag "Title of narrow")
                (repeat (symbol :tag "compile-multi types matching narrowing config"))))
  :group 'consult-compile-multi)
