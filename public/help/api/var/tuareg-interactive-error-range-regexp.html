(defconst tuareg-interactive-error-range-regexp
  (rx (* (in "\t "))
      (? "Line" (? "s") " "
         (group-n 1 (+ (in "0-9")))     ; starting line
         (? "-"
            (group-n 2 (+ (in "0-9")))) ; ending line
         ", ")
      (in "Cc") "haracters "
      (group-n 3 (+ (in "0-9")))        ; starting character
      "-"
      (group-n 4 (+ (in "0-9")))        ; ending character
      ":\n")
