(defconst majutsu-file--completion-template
  (let ((sep (format "\"%s\"" majutsu-file--completion-field-separator)))
    (concat
     (string-join
      (list "path"
            sep "file_type"
            sep "if(executable, \"t\", \"\")"
            sep "if(conflict, \"t\", \"\")")
      " ++ ")
     " ++ \"\\n\""))
