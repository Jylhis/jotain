(defcustom web-mode-auto-quote-style 1
  "Auto-quoting style."
  :group 'web-mode
  :type '(choice (const :tag "Auto-quotes with double quote" 1)
          (const :tag "Auto-quotes with single quote" 2)
          (const :tag "Auto-quotes with paren (for jsx)" 3)))
