;;;;;; Completion-at-point interface
(defconst org-roam-bracket-completion-re
  (rx "[["
      (group (opt "roam:"))
      ;; Change from ‘org-link-bracket-re’: allow empty URI part
      (group (zero-or-more
              (or (not (any "[]\\"))
                  (and "\\" (zero-or-more "\\\\") (any "[]"))
                  (and (one-or-more "\\") (not (any "[]"))))))
      "]]")
