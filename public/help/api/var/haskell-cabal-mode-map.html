    (define-key map (kbd "C-M-n") 'haskell-cabal-next-section)

    (define-key map (kbd "M-g l") 'haskell-cabal-goto-library-section)
    (define-key map (kbd "M-g e") 'haskell-cabal-goto-executable-section)
