Hidden buffer source for consult-buffer.

The source is hidden by default and can be summoned via its narrow key. All buffers are taken into account, i.e., the entire buffer-list from all frames.

(defvar consult-source-hidden-buffer
  `( :name     "Hidden Buffer"
     :narrow   ?\s
     :hidden   t
     :category buffer
     :face     consult-buffer
     :history  buffer-name-history
     :action   ,#'consult--buffer-action
     :items
     ,(lambda () (consult--buffer-query :sort 'visibility
                                        :filter 'invert
                                        :as #'consult--buffer-pair
                                        :buffer-list t)))
